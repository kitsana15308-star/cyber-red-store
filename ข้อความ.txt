                    email: email, 
                    credit: 0, 
                    role: email === 'admin@shop.com' ? 'admin' : 'user',

        window.buyProduct = async (id, price) => {
            if (!currentUser) {
                openModal('login-modal');
                return showToast("แจ้งเตือน", "กรุณาเข้าสู่ระบบก่อนซื้อสินค้า", "warning");
            }
            const userRef = doc(db, "users", currentUser.uid);
            const userSnap = await getDoc(userRef);
            const credit = userSnap.data() ? (userSnap.data().credit || 0) : 0;

            if (credit < price) return showToast("ยอดเงินไม่พอ", "เครดิตของคุณไม่พอ กรุณาเติมเงิน!", "warning");
            if (!confirm(`ยืนยันการซื้อสินค้าราคา ${price} บาท?`)) return;

            try {
                const productRef = doc(db, "products", id);
                const productSnap = await getDoc(productRef);
                const pData = productSnap.data();

                if (!pData.accounts || pData.accounts.length === 0) {
                    showToast("สินค้าหมด", "สินค้าหมดสต็อกแล้ว!", "warning");
                    loadProducts();
                    return;
                }

                const grantedAccount = pData.accounts[0];
                const remainingAccounts = pData.accounts.slice(1);

                await updateDoc(userRef, { credit: credit - price });
                await updateDoc(productRef, { accounts: remainingAccounts });
                
                await addDoc(collection(db, "orders"), {
                    uid: currentUser.uid, 
                    email: currentUser.email,
                    title: pData.title,
                    account: grantedAccount, 
                    price, 
                    purchasedAt: serverTimestamp()
                });

                showToast("สั่งซื้อสำเร็จ!", `รับข้อมูลไอดี: ${grantedAccount}`, "success");
                loadUserData(currentUser.uid);
                loadProducts();
                loadHistory();
                if(currentUser.email === 'admin@shop.com') loadAdminData();
            } catch (err) { showToast("ข้อผิดพลาด", err.message, "warning"); }
        };

        async function loadAdminData() {
            const adminOrdersList = document.getElementById('admin-orders-list');
            const adminProductsList = document.getElementById('admin-products-list');
            const totalProductsEl = document.getElementById('admin-total-products');
            const adminSlipsList = document.getElementById('admin-slips-list');
            const pendingCountEl = document.getElementById('pending-slips-count');
            const statsEl = document.getElementById('admin-stats');
            if(!adminOrdersList) return;

            const usersSnap = await getDocs(collection(db, "users"));
            cachedMembers = [];
            usersSnap.forEach(d => cachedMembers.push(d.data()));
            filterAdminMembers();

            const slipsSnap = await getDocs(collection(db, "slips"));
            let slipsHtml = '';
            let pendingCount = 0;

            slipsSnap.forEach((docSnap) => {
                const s = docSnap.data();
                if (s.status === 'pending') {
                    pendingCount++;
                    slipsHtml += `
                        <div class="bg-slate-900 border border-red-900/50 p-3 rounded-xl text-xs space-y-2">
                            <div class="flex justify-between items-center border-b border-red-950 pb-1.5">
                                <span class="text-gray-300 font-bold">${s.email}</span>
                                <span class="text-yellow-400 font-extrabold text-sm">${s.amount} ฿</span>
                            </div>
                            <div class="flex justify-between items-center text-[11px] text-gray-400">
                                <span><i class="fa-solid fa-building-columns text-red-400 mr-1"></i>โอนจาก: ${s.sender}</span>
                                <button onclick="viewSlipImage('${s.slipUrl}')" class="text-red-400 underline hover:text-white font-medium">
                                    <i class="fa-solid fa-image mr-1"></i>ดูรูปสลิป
                                </button>
                            </div>
                            <div class="flex space-x-2 pt-1">
                                <button onclick="approveSlip('${docSnap.id}', '${s.uid}', ${s.amount}, '${s.email}')" class="w-1/2 bg-emerald-600 hover:bg-emerald-700 text-white py-1.5 rounded-lg text-[11px] font-bold shadow transition-all">
                                    <i class="fa-solid fa-check mr-1"></i>อนุมัติ
                                </button>
                                <button onclick="rejectSlip('${docSnap.id}')" class="w-1/2 bg-rose-950 hover:bg-rose-900 border border-rose-800/40 text-rose-300 py-1.5 rounded-lg text-[11px] font-bold transition-all">
                                    <i class="fa-solid fa-xmark mr-1"></i>ปฏิเสธ
                                </button>
                            </div>
                        </div>
                    `;
                }
            });
            pendingCountEl.innerText = `${pendingCount} รายการ`;
            adminSlipsList.innerHTML = slipsHtml || `<p class="text-[11px] text-gray-500 text-center py-4">ไม่มีสลิปรอการตรวจสอบ</p>`;

            const ordersSnap = await getDocs(collection(db, "orders"));
            let ordersHtml = '';
            let totalSales = 0;

            ordersSnap.forEach((docSnap) => {
                const o = docSnap.data();
                totalSales += (o.price || 0);
                ordersHtml += `
                    <div class="bg-slate-950/80 border border-red-950 p-2.5 rounded-xl text-[11px] flex justify-between items-center">
                        <div>
                            <p class="text-white font-semibold">${o.title} <span class="text-gray-400 font-normal">(${o.email ? o.email.split('@')[0] : 'user'})</span></p>
                            <p class="text-[10px] text-red-300 font-mono">ID: ${o.account}</p>
                        </div>
                        <span class="text-red-400 font-bold">${o.price} ฿</span>
                    </div>
                `;
            });
            statsEl.innerText = `ยอดขายรวม: ${totalSales} ฿`;
            adminOrdersList.innerHTML = ordersHtml || `<p class="text-[11px] text-gray-500 text-center py-2">ยังไม่มีรายการซื้อในระบบ</p>`;

            const productsSnap = await getDocs(collection(db, "products"));
            let prodHtml = '';
            let prodCount = 0;

            productsSnap.forEach((docSnap) => {
                prodCount++;
                const p = docSnap.data();
                const stockCount = p.accounts ? p.accounts.length : 0;
                prodHtml += `
                    <div class="bg-slate-950/80 border border-red-950 p-2.5 rounded-xl text-[11px] flex justify-between items-center">
                        <div class="flex items-center space-x-2.5">
                            <img src="${p.imageUrl || 'https://images.unsplash.com/photo-1542751371-adc38448a05e?w=500&auto=format&fit=crop&q=60'}" class="w-10 h-10 rounded-lg object-cover border border-red-900">
                            <div>
                                <p class="text-white font-semibold line-clamp-1">${p.title} <span class="text-red-400">(${p.game})</span></p>
                                <p class="text-[10px] text-emerald-400">ราคา: ${p.price} ฿ | สต็อกคงเหลือ: ${stockCount} ไอดี</p>
                            </div>
                        </div>
                        <div class="flex space-x-1 shrink-0">
                            <button onclick="openEditProductModal('${docSnap.id}')" class="bg-sky-950 hover:bg-sky-900 text-sky-300 border border-sky-800/40 px-2 py-1 rounded-lg font-medium text-[10px]"><i class="fa-solid fa-pen"></i> แก้ไข</button>
                            <button onclick="deleteProduct('${docSnap.id}')" class="bg-red-950 hover:bg-red-900 text-red-300 border border-red-800/40 px-2 py-1 rounded-lg font-medium text-[10px]"><i class="fa-solid fa-trash"></i> ลบ</button>
                        </div>
                    </div>
                `;
            });
            totalProductsEl.innerText = `${prodCount} รายการ`;
            adminProductsList.innerHTML = prodHtml || `<p class="text-[11px] text-gray-500 text-center py-2">ยังไม่มีสินค้าในสต็อก</p>`;
        }
